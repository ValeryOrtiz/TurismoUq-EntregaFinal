  2      FUNCTION libre(p_hab NUMBER) RETURN BOOLEAN IS
  3          n NUMBER;
  4      BEGIN
  5          SELECT COUNT(*) INTO n
  6            FROM RESERVA_HABITACION rh JOIN RESERVA r ON r.reserva_id = rh.reserva_id
  7           WHERE rh.habitacion_id = p_hab AND r.estado <> 'CANCELADA'
  8             AND rh.fecha_checkin < DATE '2026-12-15' AND rh.fecha_checkout > DATE '2026-11-20';
  9          RETURN n = 0;
 10      END;
 11  BEGIN
 12      :hab_libre := NULL;
 13      :hab2      := NULL;
 14      :hab_otro  := NULL;
 15      FOR h IN (SELECT habitacion_id, capacidad_maxima, alojamiento_id
 16                  FROM HABITACION ORDER BY habitacion_id) LOOP
 17          IF :hab_libre IS NULL THEN
 18              IF h.capacidad_maxima >= 2 AND libre(h.habitacion_id) THEN
 19                  :hab_libre := h.habitacion_id; :cap := h.capacidad_maxima; :aloj_libre := h.alojamiento_id;

 22              IF libre(h.habitacion_id) THEN :hab2 := h.habitacion_id; END IF;

 24              IF libre(h.habitacion_id) THEN :hab_otro := h.habitacion_id; END IF;
